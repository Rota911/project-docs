# Account

Account management and operations.

```lua
local AccountService = require 'core.account'
```

```lua
---@class BankAccountCreateParams
---@field ownerCharId number
---@field type? BankAccountType -- 'personal' | 'company' | 'shared'
---@field label? string

---@alias BankAccountStatus 'active' | 'locked' | 'frozen' | 'closed'
```

## Create

```lua
---@param params BankAccountCreateParams
---@return BankAccount? account
---@return string? error
AccountService:Create(params)
```

```lua
local account, err = AccountService:Create({ ownerCharId = 123, type = 'personal', label = 'Main Account' })
```

## Get / GetByNumber / GetByChar / Resolve

```lua
---@param accountId number
---@return BankAccount?
AccountService:Get(accountId)

---@param accountNumber string
---@return BankAccount?
AccountService:GetByNumber(accountNumber)

---@param charId number
---@return BankAccount[]
AccountService:GetByChar(charId)

---@param identifier number|string -- accountId or accountNumber
---@return BankAccount?
AccountService:Resolve(identifier)
```

```lua
local account = AccountService:Get(5)
local account2 = AccountService:GetByNumber('LS12345678')
local accounts = AccountService:GetByChar(123)
local resolved = AccountService:Resolve('LS12345678') -- or AccountService:Resolve(42)
```

## SetStatus

```lua
---@param accountId number
---@param newStatus BankAccountStatus
---@param reason? string
---@return boolean success
---@return string? error
AccountService:SetStatus(accountId, newStatus, reason)
```

```lua
AccountService:SetStatus(5, 'frozen', 'Suspicious activity')
```

## Close

```lua
---@param accountId number
---@param reason? string
---@return boolean success
---@return string? error
AccountService:Close(accountId, reason)
```

> Account must have zero balance before closing.

```lua
---@param accountId number
```

## BankAccount Methods

```lua
local account = AccountService:Get(5)
```

### Money movement

```lua
---@param amount number
---@param reason? string
---@param srcCharId? number
---@return boolean success
---@return string? error
account:Deposit(amount, reason, srcCharId)
account:Withdraw(amount, reason, srcCharId)

---@param targetAccountId number
---@param amount number
---@param reason? string
---@return boolean success
---@return string? error
account:Transfer(targetAccountId, amount, reason)

---@return number
account:GetBalance()
```

```lua
account:Deposit(5000, 'Initial deposit')
account:Transfer(42, 1000, 'Payment')
```

### Cards

```lua
---@return BankCard[]
account:GetCards()

---@param charId number
---@param cardType? BankCardType -- 'debit' | 'credit'
---@return BankCard? card
---@return string? error
account:CreateCard(charId, cardType)
```

```lua
local card, err = account:CreateCard(123, 'debit')
```

### GetTransactions

```lua
---@param limit? number
---@param offset? number
---@return BankTransactionData[]
account:GetTransactions(limit, offset)
```

```lua
local history = account:GetTransactions(20, 0)
```

### Permissions & members

```lua
---@param charId number
---@param permission BankPermission
---@return boolean
account:HasPermission(charId, permission)

---@param charId number
---@param permission BankPermission
---@return boolean success
---@return string? error
account:GrantPermission(charId, permission)
account:RevokePermission(charId, permission)

---@return BankAccountMember[]
account:GetMembers()
```

### Close

```lua
---@param reason? string
---@return boolean success
---@return string? error
account:Close(reason)
```

> Requires zero balance, same rule as `AccountService:Close`.
