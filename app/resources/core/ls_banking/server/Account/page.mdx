# Account

Account management and operations.

```lua
local AccountService = require 'core.account'
```

## Create

```lua
---@param params BankAccountCreateParams
---@return BankAccount? account
---@return string? error
AccountService:Create(params)
```

```lua
---@class BankAccountCreateParams
---@field ownerCharId number
---@field type? BankAccountType -- 'personal' | 'company' | 'shared'
---@field label? string
```

```lua
local account, err = AccountService:Create({
    ownerCharId = 123,
    type = 'personal',
    label = 'Main Account'
})
```

## Get

```lua
---@param accountId number
---@return BankAccount?
AccountService:Get(accountId)
```

```lua
local account = AccountService:Get(5)
if account then print(account.balance) end
```

## GetByNumber

```lua
---@param accountNumber string
---@return BankAccount?
AccountService:GetByNumber(accountNumber)
```

```lua
local account = AccountService:GetByNumber('LS12345678')
```

## GetByChar

```lua
---@param charId number
---@return BankAccount[]
AccountService:GetByChar(charId)
```

```lua
local accounts = AccountService:GetByChar(123)
for i = 1, #accounts do
    print(accounts[i].accountNumber, accounts[i].balance)
end
```

## Resolve

```lua
---@param identifier number|string -- accountId or accountNumber
---@return BankAccount?
AccountService:Resolve(identifier)
```

```lua
local account = AccountService:Resolve('LS12345678')
local account2 = AccountService:Resolve(42)
```

## SetStatus

```lua
---@param accountId number
---@param newStatus BankAccountStatus
---@param reason? string
---@return boolean success
---@return string? error
AccountService:SetStatus(accountId, newStatus, reason)
```

```lua
---@alias BankAccountStatus 'active' | 'locked' | 'frozen' | 'closed'
```

```lua
AccountService:SetStatus(5, 'frozen', 'Suspicious activity')
```

## Close

```lua
---@param accountId number
---@param reason? string
---@return boolean success
---@return string? error
AccountService:Close(accountId, reason)
```

> Account must have zero balance before closing.

## SetLabel

```lua
---@param accountId number
---@param label string
---@return boolean success

AccountService:SetLabel(accountId, label)
```
